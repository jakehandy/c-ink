cmake_minimum_required(VERSION 3.16)
project(cink_sim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_subdirectory(../firmware/components/cink_core cink_core)
add_executable(cink_sim main.cpp)
target_link_libraries(cink_sim PRIVATE cink_core)
add_executable(cink_tests tests.cpp)
target_link_libraries(cink_tests PRIVATE cink_core)
enable_testing()
add_test(NAME cink_tests COMMAND cink_tests)
